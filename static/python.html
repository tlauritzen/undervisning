Løkker

Løkker kan løbe igennem data og lave den samme operation på alle elementer
while 
							
								antal_aar = 5
								aar = 0
								hovedstol = 143000
								rente = 0.03
								while aar < antal_aar:
									hovedstol *= 1 + rente
									aar += 1

								print(hovedstol)
for 
							
								antal_aar = 5
								aar = 0
								hovedstol = 143000
								rente = 0.03
								for arr in range(antal_aar):
									hovedstol *= 1 + rente

								print(hovedstol)

Betinget udførelse (Conditional Execution)

for 
							
								if gaeld > hovedstol * antal_aar * rente:

								elif gaeld < hovedstol * antal_aar * rente:
									print("Underligt...")

									print("Heller ikke normalt")
